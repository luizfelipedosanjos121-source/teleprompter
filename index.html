// 📊 ID DA PLANILHA const SHEET_ID = 'SEU_ID_DA_PLANILHA_AQUI'; // Mude para o ID da sua planilha const SHEET_NAME = 'Músicas'; // Nome da aba // 🎵 OBTER TODAS AS MÚSICAS function obterMusicas() { try { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const dados = sheet.getDataRange().getValues(); const musicas = []; // Pula o cabeçalho (linha 1) for (let i = 1; i < dados.length; i++) { const [id, nome, artista, letrasTexto, dataCriacao] = dados[i]; if (nome && artista && letrasTexto) { musicas.push({ id: id || i, nome: nome, artista: artista, letras: letrasTexto.split('\n').filter(l => l.trim() !== ''), dataCriacao: dataCriacao || new Date().toLocaleDateString('pt-BR') }); } } return musicas; } catch (erro) { Logger.log('Erro ao obter músicas: ' + erro); return []; } } // ➕ ADICIONAR MÚSICA function adicionarMusica(nome, artista, letras) { try { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); // Validar if (!nome || !artista || !letras) { return { sucesso: false, mensagem: 'Campos obrigatórios não preenchidos' }; } const id = Date.now(); // ID único baseado em timestamp const dataCriacao = new Date().toLocaleDateString('pt-BR'); // Adicionar linha sheet.appendRow([ id, nome, artista, letras, dataCriacao ]); return { sucesso: true, mensagem: 'Música adicionada com sucesso!' }; } catch (erro) { return { sucesso: false, mensagem: 'Erro: ' + erro.toString() }; } } // 🗑️ DELETAR MÚSICA function deletarMusica(id) { try { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (dados[i][0] == id) { sheet.deleteRow(i + 1); return { sucesso: true, mensagem: 'Música deletada com sucesso!' }; } } return { sucesso: false, mensagem: 'Música não encontrada' }; } catch (erro) { return { sucesso: false, mensagem: 'Erro: ' + erro.toString() }; } } // ✏️ ATUALIZAR MÚSICA function atualizarMusica(id, nome, artista, letras) { try { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (dados[i][0] == id) { sheet.getRange(i + 1, 2, 1, 3).setValues([[nome, artista, letras]]); return { sucesso: true, mensagem: 'Música atualizada com sucesso!' }; } } return { sucesso: false, mensagem: 'Música não encontrada' }; } catch (erro) { return { sucesso: false, mensagem: 'Erro: ' + erro.toString() }; } } // 📤 EXPORTAR PARA WEB (via HTTP) function doGet(e) { try { const acao = e.parameter.acao; if (acao === 'obter') { const musicas = obterMusicas(); return ContentService.createTextOutput(JSON.stringify({ sucesso: true, musicas: musicas })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ sucesso: false, mensagem: 'Ação não reconhecida' })).setMimeType(ContentService.MimeType.JSON); } catch (erro) { return ContentService.createTextOutput(JSON.stringify({ sucesso: false, mensagem: 'Erro: ' + erro.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { return doGet(e); } // 🏗️ INICIALIZAR PLANILHA (executar uma vez) function inicializarPlanilha() { try { const ss = SpreadsheetApp.create('Teleprompter - Músicas'); const sheet = ss.getActiveSheet(); sheet.appendRow(['ID', 'Nome', 'Artista', 'Letras', 'Data Criação']); // Exemplo de música sheet.appendRow([ Date.now(), 'Imagine', 'John Lennon', 'Imagine all the people\nLiving life in peace\nYou may say I\'m a dreamer\nBut I\'m not the only one', new Date().toLocaleDateString('pt-BR') ]); Logger.log('Planilha criada: ' + ss.getUrl()); alert('✅ Planilha criada! ID: ' + ss.getId()); } catch (erro) { Logger.log('Erro: ' + erro); } }
